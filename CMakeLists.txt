cmake_minimum_required(VERSION 3.22)
project(StudioSP VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(JUCE_SOURCE_DIR "" CACHE PATH "Optional local JUCE 8 checkout")
if(JUCE_SOURCE_DIR)
    add_subdirectory("${JUCE_SOURCE_DIR}" juce)
else()
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
set(RUBBERBAND_SOURCE_DIR "" CACHE PATH "Optional local Rubber Band 3.3.0 checkout")
if(NOT RUBBERBAND_SOURCE_DIR)
    FetchContent_Declare(RubberBand
        GIT_REPOSITORY https://github.com/breakfastquay/rubberband.git
        GIT_TAG 2be46b0dffb13273a67396c77bc9278736bb03d2)
    FetchContent_MakeAvailable(RubberBand)
    set(RUBBERBAND_SOURCE_DIR "${rubberband_SOURCE_DIR}")
endif()
add_library(StudioSPStretch STATIC "${RUBBERBAND_SOURCE_DIR}/single/RubberBandSingle.cpp")
target_include_directories(StudioSPStretch PUBLIC "${RUBBERBAND_SOURCE_DIR}")
target_compile_definitions(StudioSPStretch PRIVATE NOMINMAX)
set_target_properties(StudioSPStretch PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
target_link_libraries(StudioSPStretch PRIVATE Threads::Threads)
if(APPLE)
    target_link_libraries(StudioSPStretch PRIVATE "-framework Accelerate")
endif()
set(STUDIOSP_FORMATS Standalone VST3)
if(APPLE)
    list(APPEND STUDIOSP_FORMATS AU)
endif()
juce_add_plugin(StudioSP COMPANY_NAME "StudioSP Audio" BUNDLE_ID "audio.studiosp.sampler"
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
    PLUGIN_MANUFACTURER_CODE StSP PLUGIN_CODE Sps1 FORMATS ${STUDIOSP_FORMATS} PRODUCT_NAME "StudioSP")
target_sources(StudioSP PRIVATE src/main.cpp src/ui/MainComponent.cpp)
juce_add_binary_data(StudioSPBrand NAMESPACE StudioSPBrand HEADER_NAME StudioSPBrand.h
    SOURCES assets/branding/StudioSPLogo.png)
set_target_properties(StudioSPBrand PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(StudioSP PRIVATE StudioSPBrand)
target_include_directories(StudioSP PRIVATE src)
target_compile_definitions(StudioSP PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(StudioSP PRIVATE juce::juce_audio_utils juce::juce_dsp
    PUBLIC StudioSPStretch
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
include(CTest)
if(BUILD_TESTING)
    add_executable(StudioSPTests tests/DspTests.cpp tests/MidiTests.cpp)
    target_include_directories(StudioSPTests PRIVATE src)
    target_link_libraries(StudioSPTests PRIVATE StudioSP juce::juce_audio_basics juce::juce_recommended_config_flags)
    add_test(NAME StudioSP.DSP COMMAND StudioSPTests)
endif()
